ls -li                         # 5. xem inode — copy kết quả vào ex3.txt
# 1. Gộp /bin và /usr/bin, sort, bỏ trùng, đếm số lượng
ls /bin /usr/bin | sort | uniq | wc -l

# 2. Lọc ra tên có chứa "zip"
ls /bin /usr/bin | sort | uniq | grep zip

# 3. Tạo file text ≥30 dòng, ví dụ:
seq 1 30 > mytext.txt          # hoặc dùng nano/vim để tự viết nội dung
head -n 5 mytext.txt           # 5 dòng đầu
tail -n 5 mytext.txt           # 5 dòng cuối
wc mytext.txt                  # dòng, từ, byte (wc -l -w -c)

# 4. Liệt kê thư mục, lưu ra file + hiện trên màn hình

mkdir permissions

touch secret.txt shared.txt public.txt

chmod 600 secret.txt    # rw------- : chỉ owner đọc/ghi
chmod 664 shared.txt    # rw-rw-r-- : owner+group đọc/ghi, other chỉ đọc
chmod 777 public.txt    # rwxrwxrwx : mọi người đọc/ghi/thực thi
